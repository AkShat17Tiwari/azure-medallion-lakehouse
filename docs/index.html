-- 1. Daily Orders & Delivery SLA Metrics
SELECT 
    order_date,
    order_status,
    total_orders,
    unique_customers,
    ROUND(avg_delivery_days, 1) AS avg_delivery_days,
    delivered_orders
FROM public.gold_daily_orders_summary
WHERE order_date >= '2026-01-01'
ORDER BY order_date DESC, total_orders DESC;
-- 2. Top Revenue Pickup Zones (NYC Taxi Mart)
SELECT 
    pulocationid AS pickup_zone,
    SUM(total_trips) AS total_trips,
    ROUND(SUM(total_revenue), 2) AS total_revenue,
    ROUND(AVG(avg_fare_per_trip), 2) AS avg_fare,
    ROUND(AVG(avg_tip_amount), 2) AS avg_tip
FROM public.gold_daily_zone_metrics
GROUP BY pulocationid
ORDER BY total_revenue DESC
LIMIT 10;
